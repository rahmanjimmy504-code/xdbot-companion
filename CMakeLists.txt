cmake_minimum_required(VERSION 3.25)
project(XdBotCompanion VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
add_library(${PROJECT_NAME} SHARED src/main.cpp src/MacroScanner.cpp src/AndroidShare.cpp src/MacroPicker.cpp)
if(NOT DEFINED ENV{GEODE_SDK})
    message(FATAL_ERROR "Set GEODE_SDK to a compatible Geode v5.10.1 SDK")
endif()
add_subdirectory($ENV{GEODE_SDK} ${CMAKE_CURRENT_BINARY_DIR}/geode)
setup_geode_mod(${PROJECT_NAME} DONT_INSTALL)
