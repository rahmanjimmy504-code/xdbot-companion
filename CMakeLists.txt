cmake_minimum_required(VERSION 3.25)
project(XdBotCompanion VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(SHARE_BACKEND "")
set(HYPERBOLUS_BACKEND "")
set(AI_BACKEND "")
set(EXPORT_BACKEND src/MacroExport.cpp)
if(GEODE_TARGET_PLATFORM STREQUAL "Android64" OR GEODE_TARGET_PLATFORM STREQUAL "Android32")
    set(SHARE_BACKEND src/AndroidShare.cpp)
    set(HYPERBOLUS_BACKEND src/HyperbolusAndroid.cpp)
    set(AI_BACKEND src/ai/AiGameplayAndroid.cpp)
elseif(GEODE_TARGET_PLATFORM STREQUAL "Win64")
    set(SHARE_BACKEND src/WindowsShare.cpp)
elseif(GEODE_TARGET_PLATFORM STREQUAL "iOS")
    enable_language(OBJCXX)
    set(SHARE_BACKEND src/IOSShare.mm)
    set(EXPORT_BACKEND src/IOSMacroExport.mm)
else()
    message(FATAL_ERROR "Only Android32, Android64, Win64 and iOS are in scope")
endif()
# Experimental local AI heuristic has an explicit Android pause-menu button,
# but native beta.19 CI and device behavior are unverified; not a general solver.
add_library(${PROJECT_NAME} SHARED src/main.cpp src/MacroScanner.cpp src/ShareMacros.cpp src/MacroExportCore.cpp src/MacroPicker.cpp src/ai/AiMacroSearch.cpp src/ai/AiReplayJson.cpp ${AI_BACKEND} ${EXPORT_BACKEND} ${SHARE_BACKEND} ${HYPERBOLUS_BACKEND})
if(GEODE_TARGET_PLATFORM STREQUAL "Win64")
    target_link_libraries(${PROJECT_NAME} windowsapp)
endif()
if(GEODE_TARGET_PLATFORM STREQUAL "iOS")
    set_source_files_properties(src/IOSShare.mm src/IOSMacroExport.mm PROPERTIES COMPILE_FLAGS "-fobjc-arc")
    target_link_libraries(${PROJECT_NAME} "-framework UIKit" "-framework Foundation")
endif()
if(NOT DEFINED ENV{GEODE_SDK})
    message(FATAL_ERROR "Set GEODE_SDK to a compatible Geode v5.10.1 SDK")
endif()
add_subdirectory($ENV{GEODE_SDK} ${CMAKE_CURRENT_BINARY_DIR}/geode)
setup_geode_mod(${PROJECT_NAME} DONT_INSTALL)
